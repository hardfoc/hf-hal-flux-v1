66constexpr const char* TemperatureErrorToString(TemperatureError error) noexcept {
67 switch (error) {
68 case TemperatureError::SUCCESS: return "Success";
69 case TemperatureError::NOT_INITIALIZED: return "Not initialized";
70 case TemperatureError::INITIALIZATION_FAILED: return "Initialization failed";
71 case TemperatureError::SENSOR_NOT_FOUND: return "Sensor not found";
72 case TemperatureError::SENSOR_NOT_AVAILABLE: return "Sensor not available";
73 case TemperatureError::READ_FAILED: return "Read failed";
74 case TemperatureError::INVALID_PARAMETER: return "Invalid parameter";
75 case TemperatureError::NO_SENSORS_AVAILABLE: return "No sensors available";
76 case TemperatureError::ALLOCATION_FAILED: return "Allocation failed";
77 case TemperatureError::DEPENDENCY_NOT_READY: return "Dependency not ready";
78 case TemperatureError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
