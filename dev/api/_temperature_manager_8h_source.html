65constexpr const char* TemperatureErrorToString(TemperatureError error) noexcept {
66 switch (error) {
67 case TemperatureError::SUCCESS: return "Success";
68 case TemperatureError::NOT_INITIALIZED: return "Not initialized";
69 case TemperatureError::INITIALIZATION_FAILED: return "Initialization failed";
70 case TemperatureError::SENSOR_NOT_FOUND: return "Sensor not found";
71 case TemperatureError::SENSOR_NOT_AVAILABLE: return "Sensor not available";
72 case TemperatureError::READ_FAILED: return "Read failed";
73 case TemperatureError::INVALID_PARAMETER: return "Invalid parameter";
74 case TemperatureError::NO_SENSORS_AVAILABLE: return "No sensors available";
75 case TemperatureError::ALLOCATION_FAILED: return "Allocation failed";
76 case TemperatureError::DEPENDENCY_NOT_READY: return "Dependency not ready";
