69constexpr const char* CommErrorToString(CommError error) noexcept {
70 switch (error) {
71 case CommError::SUCCESS: return "Success";
72 case CommError::NOT_INITIALIZED: return "Not initialized";
73 case CommError::INITIALIZATION_FAILED: return "Initialization failed";
74 case CommError::BUS_NOT_AVAILABLE: return "Bus not available";
75 case CommError::DEVICE_NOT_FOUND: return "Device not found";
76 case CommError::INVALID_BUS_INDEX: return "Invalid bus index";
77 case CommError::INVALID_DEVICE_INDEX: return "Invalid device index";
78 case CommError::DEVICE_ALREADY_EXISTS: return "Device already exists";
79 case CommError::DEVICE_CREATION_FAILED: return "Device creation failed";
80 case CommError::COMMUNICATION_FAILED: return "Communication failed";
81 case CommError::INVALID_PARAMETER: return "Invalid parameter";
82 case CommError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
83 default: return "Unknown error";
84 }
85}
