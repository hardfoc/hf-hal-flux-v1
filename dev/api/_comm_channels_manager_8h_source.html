77constexpr const char* CommErrorToString(CommError error) noexcept {
78 switch (error) {
79 case CommError::SUCCESS: return "Success";
80 case CommError::NOT_INITIALIZED: return "Not initialized";
81 case CommError::INITIALIZATION_FAILED: return "Initialization failed";
82 case CommError::BUS_NOT_AVAILABLE: return "Bus not available";
83 case CommError::DEVICE_NOT_FOUND: return "Device not found";
84 case CommError::INVALID_BUS_INDEX: return "Invalid bus index";
85 case CommError::INVALID_DEVICE_INDEX: return "Invalid device index";
86 case CommError::DEVICE_ALREADY_EXISTS: return "Device already exists";
87 case CommError::DEVICE_CREATION_FAILED: return "Device creation failed";
88 case CommError::COMMUNICATION_FAILED: return "Communication failed";
89 case CommError::INVALID_PARAMETER: return "Invalid parameter";
90 case CommError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
91 default: return "Unknown error";
92 }
93}
