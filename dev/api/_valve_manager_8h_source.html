591 class Max22200Access {
592 public:
594 explicit operator bool() const noexcept { return handler_ != nullptr; }
595
597 max22200::DriverStatus GetStatus(max22200::StatusConfig& status) noexcept {
598 return handler_ ? handler_->GetStatus(status)
599 : max22200::DriverStatus::INITIALIZATION_ERROR;
600 }
601
603 max22200::DriverStatus ConfigureChannel(uint8_t channel,
604 const max22200::ChannelConfig& config) noexcept {
605 return handler_ ? handler_->ConfigureChannel(channel, config)
606 : max22200::DriverStatus::INITIALIZATION_ERROR;
607 }
608
614 template <typename Fn>
615 auto WithDriver(Fn&& fn) noexcept {
616 using R = std::invoke_result_t<Fn, Max22200Handler::DriverType&>;
617 if (handler_ == nullptr) {
618 if constexpr (std::is_void_v<R>) {
619 return false;
620 } else if constexpr (std::is_same_v<R, max22200::DriverStatus>) {
621 return max22200::DriverStatus::INITIALIZATION_ERROR;
622 } else {
623 return R{};
624 }
625 }
626 return handler_->WithDriver(std::forward<Fn>(fn));
627 }
628
629 private:
630 friend class ValveManager;
631 explicit Max22200Access(Max22200Handler* handler) noexcept : handler_(handler) {}
632 Max22200Handler* handler_;
633 };
