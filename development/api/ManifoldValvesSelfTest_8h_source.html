71 //--------------------------------------------------------------------
72 // Proportional step-response characterization
73 //--------------------------------------------------------------------
74 //
75 // Each step of the 0 → full → 0 ladder is exercised in two phases:
76 //
77 // 1) Settling: poll the chip's FB_I_AVG every `prop_sample_period_ms`
78 // until the actual current stays within `±settling_band` for
79 // `settling_consec_samples` consecutive reads, or until
80 // `settling_max_ms` elapses (timeout = "did not settle"). The
81 // worst |err| seen during this phase is recorded as
82 // `overshoot_max_mA`.
83 //
84 // 2) Steady-state: hold the setpoint for `steady_state_ms` and
85 // accumulate sum / sum_of_squares / max-|err|. Compute
86 // mean error and standard deviation; check both against
87 // `accuracy_band_*` and `noise_floor_mA`.
88 //
89 // This separates the valve's *dynamic* response (settling time +
90 // overshoot — physics) from its *static* accuracy (mean error + noise —
91 // chip + sensor). LM-Pro settles in ~30 ms, slow VSO solenoids in
92 // ~80–200 ms; using a single fixed snapshot would either fail on the
93 // slow ones (snapshot mid-transient) or paper over real settling
94 // problems (snapshot way after settling). Two-phase accounting catches
95 // both classes of failure cleanly.

162 //--------------------------------------------------------------------
163 // All-proportionals parallel stress test (worst-case driver load)
164 //--------------------------------------------------------------------
165 //
166 // Drives every TLE92466ED channel simultaneously through a synchronized
167 // step ladder. Catches what a per-valve sweep cannot:
168 // - VBAT droop under simultaneous high current draw (sum across
169 // all 6 channels can exceed 600 mA, ~14 W into the load),
170 // - SPI saturation (every dirty Apply pushes a setpoint write),
171 // - Central thermal warnings (`tle_ot_warning`),
172 // - Per-channel current-regulation / pwm-regulation warnings that
173 // only show up when the PSU can't keep up,
174 // - Driver state-machine races between Configuration-Mode round-trips
175 // happening on different channels.
176 //
177 // Two patterns are run back-to-back:
178 // 1. UNIFORM — every channel at the same normalized position; walks
179 // 0 → 33 → 66 → 100 → 66 → 33 → 0. This is the
180 // worst-case current-draw scenario.
181 // 2. STAGGERED — at each step, channel i gets level[(step + i) %
182 // num_levels]. After num_levels steps every channel
183 // has visited every level; tests realistic phase-
184 // staggered control behaviour (different VBAT
185 // droop / regulation profile than uniform).
186 //
187 // The stress test enforces explicit pass/fail tolerances (below) and
188 // its verdict feeds into the overall self-test result. Set
189 // `parallel_stress_diagnostic_only = true` to revert to the original
190 // behaviour where the stress test only logs and never fails (useful
191 // during commissioning / first light).

193 bool parallel_stress_run_uniform{true};
194 bool parallel_stress_run_staggered{true};
195 bool parallel_stress_diagnostic_only{false};

212 //--------------------------------------------------------------------
213 // Parallel-stress PASS/FAIL tolerances
214 //--------------------------------------------------------------------
215 //
216 // For each (channel × stress level) we evaluate three sub-checks:
217 //
218 // |mean_err| ≤ max(max_abs_mean_err_min_mA,
219 // max_abs_mean_err_pct% × cmd_mA)
220 // stddev ≤ max_stddev_mA
221 // |excursion| ≤ max(max_excursion_min_mA,
222 // max_excursion_pct% × cmd_mA)
223 //
224 // A step "fails" if any sub-check fails OR if the channel's snapshot
225 // shows a hard fault flag. Per-channel pass requires that the number
226 // of failed steps is ≤ `parallel_max_failed_steps_per_channel` AND no
227 // forbidden warning flag was raised. Pattern pass requires every
228 // channel pass AND none of the subsystem fail flags below were seen.
229 // Stress test passes (and feeds into overall_pass) only when every
230 // executed pattern passes — unless `parallel_stress_diagnostic_only`
231 // is true, in which case the verdict is logged but does not lower
232 // overall_pass.
233 //
234 // Defaults are tuned for the WH bench: VSO valves typically settle
235 // within ±5 mA of setpoint under simultaneous load; LM-Pro can
236 // shortfall ~20 mA at full simultaneous current draw (494 mA total)

238 // catches "no current at all" and "wildly off" but tolerates that

417 static const SelfTestReport& Run(ManifoldValves& system,
418 ManifoldValvesDiagnosticsThread& diag,
419 const SelfTestPlan& plan = SelfTestPlan::All()) noexcept {
420 // Keep the (~3 kB) report off the main task's small stack.
421 static SelfTestReport report{};
422 report = SelfTestReport{};
423
424 ESP_LOGI(kTag, "");
425 ESP_LOGI(kTag, "════════════════════════════════════════════════════════════");
426 ESP_LOGI(kTag, " Manifold valves self-test — bench commissioning sweep");
427 ESP_LOGI(kTag, " one valve at a time, 1× ON/OFF or ramp + diagnostics");
428 ESP_LOGI(kTag, "════════════════════════════════════════════════════════════");
429 ESP_LOGI(kTag, "");
430
431 // Make sure the diagnostics thread has at least one fresh pass before
432 // we start so subsequent fault-count deltas are meaningful.
433 WaitForDiagPass(diag);
434
435 // ── Solenoid sweep ───────────────────────────────────────────────
436 for (std::size_t i = 0; i < kNumSolenoids; ++i) {
437 const auto& entry = kSolenoidMap[i];
438 auto& valve = system.Get(static_cast<SolenoidId>(i));
439
440 if (!plan.enable_solenoids[i]) {
441 ESP_LOGW(kTag, "[SOL %.*s] skipped (disabled in plan)",
442 static_cast<int>(entry.label.size()), entry.label.data());
443 report.solenoids[i].label = entry.label;
444 report.solenoids[i].channel = entry.manager_channel;
445 report.solenoids[i].pass = true; // skipped == not failed
446 continue;
447 }
448
449 PromptOperator(plan, "SOL", entry.label, entry.model.label);
450 report.solenoids[i] = TestSolenoid(valve, entry, plan, diag);
451 if (!report.solenoids[i].pass) report.overall_pass = false;
452 DwellBetweenValves(plan);
453 }
454
455 // ── Per-valve proportional sweep (optional, default OFF) ─────────
456 //
457 // Original mode: walk each TLE channel one valve at a time, ramping
458 // 0 → full → 0 with per-step tracking checks. Useful when the
459 // operator can attach a single proportional valve to the manifold.
460 // Disabled by default (`enable_per_valve_proportional = false`) on
461 // the WH bench because it's impractical to physically swap the
462 // manifold connection between every channel — the parallel stress
463 // test (below) gives equivalent coverage in one shot.
464 if (plan.enable_per_valve_proportional) {
465 for (std::size_t i = 0; i < kNumProportionals; ++i) {
466 const auto& entry = kProportionalMap[i];
467 auto& valve = system.Get(static_cast<ProportionalId>(i));
468
469 if (!plan.enable_proportionals[i]) {
470 ESP_LOGW(kTag, "[PROP %.*s] skipped (disabled in plan)",
471 static_cast<int>(entry.label.size()), entry.label.data());
472 report.proportionals[i].label = entry.label;
473 report.proportionals[i].channel = entry.manager_channel;
474 report.proportionals[i].pass = true;
475 continue;
476 }
477
478 PromptOperator(plan, "PROP", entry.label, entry.model.label);
479 report.proportionals[i] = TestProportional(valve, entry, plan, diag);
480 if (!report.proportionals[i].pass) report.overall_pass = false;
481 DwellBetweenValves(plan);
482 }
483 } else {
484 // Mark per-valve proportional rows as "skipped — parallel only"
485 // so the summary block stays consistent.
486 for (std::size_t i = 0; i < kNumProportionals; ++i) {
487 report.proportionals[i].label = kProportionalMap[i].label;
488 report.proportionals[i].channel = kProportionalMap[i].manager_channel;
489 report.proportionals[i].pass = true; // skipped == not failed
490 }
491 }
492
493 // ── All-proportionals parallel stress test (worst-case load) ────
494 if (plan.enable_parallel_stress_test) {
495 if (plan.parallel_stress_run_uniform) {
496 report.stress_uniform =
497 RunParallelStress(system, diag, plan, /*staggered=*/false);
498 }
499 if (plan.parallel_stress_run_staggered) {
500 if (plan.parallel_stress_run_uniform &&
501 plan.parallel_inter_pattern_pause_ms > 0) {
502 ESP_LOGI(kTag,
503 " …pausing %lu ms between UNIFORM and STAGGERED so VBAT recovers…",
504 static_cast<unsigned long>(plan.parallel_inter_pattern_pause_ms));
505 os_delay_msec(plan.parallel_inter_pattern_pause_ms);
506 }
507 report.stress_staggered =
508 RunParallelStress(system, diag, plan, /*staggered=*/true);
509 }
510
511 // Stress verdicts feed into overall_pass unless the operator
512 // explicitly opted into diagnostic-only mode (commissioning).
513 if (!plan.parallel_stress_diagnostic_only) {
514 if (report.stress_uniform.executed && !report.stress_uniform.pass) {
515 report.overall_pass = false;
516 }
517 if (report.stress_staggered.executed && !report.stress_staggered.pass) {
518 report.overall_pass = false;
519 }
520 }
521 }
522
523 PrintSummary(report);
524 return report;
525 }

600 //---------------------------------------------------------------------
601 // Proportional: 0 → full → 0 triangle ramp + setpoint-vs-actual tracking
602 //---------------------------------------------------------------------
603 //
604 // For each setpoint visited along the triangle we:
605 // 1. Request the position (lock-free atomic, picked up by the
606 // ManifoldValvesControlThread on its next tick).
607 // 2. Sleep `proportional_settle_ms` (covers ≥ one dither period + one
608 // diagnostic poll).
609 // 3. Wait one full diagnostics-thread sweep so the snapshot's
610 // `average_current_mA` reflects the new setpoint.
611 // 4. Compare measured |I_avg − setpoint| against
612 // `max(tracking_band_min_mA, tracking_band_pct% × full_flow)`.
613 //
614 // Up-ramp actuals must strictly increase and down-ramp actuals must
615 // strictly decrease (with a small 1 mA hysteresis to absorb measurement
616 // noise) — that's what catches "the valve doesn't actually track the
617 // setpoint" failures (stuck plunger, broken coil, EN pin LOW, …).
618 static SelfTestResult TestProportional(ProportionalValve& valve,
619 const ProportionalMapEntry& entry,
620 const SelfTestPlan& plan,
621 ManifoldValvesDiagnosticsThread& diag) noexcept {
622 SelfTestResult res{};
623 res.label = entry.label;
624 res.channel = entry.manager_channel;
625
626 const uint32_t apply_errors_before = valve.GetApplyErrorCount();
627 const uint32_t fault_count_before = valve.GetFaultCount();
628
629 const uint32_t steps = (plan.proportional_steps < 4) ? 4 : plan.proportional_steps;
630 const uint16_t full_flow_ma = entry.model.full_flow_current_ma;
631 const float band_pct = (plan.tracking_band_pct * full_flow_ma) / 100.0f;
632 const float band_mA = (band_pct > plan.tracking_band_min_mA)
633 ? band_pct : plan.tracking_band_min_mA;
634
635 ESP_LOGI(kTag,
636 "[PROP %.*s] CH=PROP_CH%u model=%.*s triangle 0→%u→0 mA "
637 "steps=%lu band=±%.1f mA",
638 static_cast<int>(entry.label.size()), entry.label.data(),
639 ChannelHwIndex(entry.manager_channel),
640 static_cast<int>(entry.model.label.size()),
641 entry.model.label.data(),
642 static_cast<unsigned>(full_flow_ma),
643 static_cast<unsigned long>(steps),
644 static_cast<double>(band_mA));
645
646 // Reset to 0 first.
647 valve.RequestPosition(0.0f);
648 os_delay_msec(80);
649 WaitForDiagPass(diag);
650
651 int32_t err_sum_mA = 0;
652 uint32_t err_count = 0;
653 int16_t max_abs_err = 0;
654 int16_t prev_actual = 0;
655 bool first_sample = true;
656
657 auto walk_and_check = [&](float pos, const char* tag, bool ascending) {
658 valve.RequestPosition(pos);
659 os_delay_msec(plan.proportional_settle_ms);
660 WaitForDiagPass(diag);
661
662 const auto snap = valve.Snapshot();
663 const int16_t actual_mA = snap.average_current_mA;
664 const int16_t cmd_mA = static_cast<int16_t>(
665 pos * static_cast<float>(full_flow_ma) + 0.5f);
666 const int16_t err_mA = static_cast<int16_t>(actual_mA - cmd_mA);
667 const int16_t abs_err_mA = static_cast<int16_t>(err_mA < 0 ? -err_mA : err_mA);
668
669 const bool inside_band =
670 static_cast<float>(abs_err_mA) <= band_mA;
671 if (!inside_band) ++res.tracking_outliers;
672 err_sum_mA += err_mA;
673 ++err_count;
674 if (abs_err_mA > max_abs_err) max_abs_err = abs_err_mA;
675
676 // Monotonicity check (1 mA hysteresis to absorb chip noise).
677 if (!first_sample) {
678 const int delta = actual_mA - prev_actual;
679 if (ascending && delta < -1) res.ramp_monotonic_up = false;
680 if (!ascending && delta > 1) res.ramp_monotonic_down = false;
681 }
682 first_sample = false;
683 prev_actual = actual_mA;
684
685 ESP_LOGI(kTag,
686 " %.*s [%s]: pos=%.2f cmd=%4d mA I_act=%5d mA "
687 "err=%+4d mA (%s) duty=%4u/10000 flags=[oc=%d sg=%d ol=%d ot=%d olsg=%d]"
688 " warn=[ot=%d cur=%d pwm=%d olsg=%d]",
689 static_cast<int>(entry.label.size()), entry.label.data(),
690 tag,
691 static_cast<double>(pos),
692 static_cast<int>(cmd_mA),
693 static_cast<int>(actual_mA),
694 static_cast<int>(err_mA),
695 inside_band ? "OK" : "OUT",
696 snap.duty_cycle_permyriad,
697 snap.overcurrent, snap.short_to_ground, snap.open_load,
698 snap.over_temperature, snap.open_load_short_ground,
699 snap.ot_warning, snap.current_regulation_warning,
700 snap.pwm_regulation_warning, snap.olsg_warning);
701 };
702
703 // Ramp UP 0 → 1
704 for (uint32_t s = 0; s <= steps; ++s) {
705 const float pos = static_cast<float>(s) / static_cast<float>(steps);
706 walk_and_check(pos, "UP ", /*ascending=*/true);
707 }
708 // Ramp DOWN 1 → 0
709 first_sample = true; // restart monotonicity check for the down-ramp
710 for (int32_t s = static_cast<int32_t>(steps) - 1; s >= 0; --s) {
711 const float pos = static_cast<float>(s) / static_cast<float>(steps);
712 walk_and_check(pos, "DOWN", /*ascending=*/false);
713 }
714
715 valve.RequestPosition(0.0f);
716 os_delay_msec(80);
717
718 res.apply_errors = valve.GetApplyErrorCount() - apply_errors_before;
719 res.fault_count = valve.GetFaultCount() - fault_count_before;
720 res.tracking_samples = err_count;
721 res.tracking_max_abs_err_mA = max_abs_err;
722 res.tracking_mean_err_mA = (err_count > 0)
723 ? static_cast<int16_t>(err_sum_mA / static_cast<int32_t>(err_count))
724 : 0;
725
726 const auto snap = valve.Snapshot();
727 const bool tracking_ok = (res.tracking_outliers <= plan.tracking_max_outliers) &&
728 res.ramp_monotonic_up && res.ramp_monotonic_down;
729 res.pass = (res.apply_errors == 0) && (res.fault_count == 0) &&
730 !snap.any_fault() && tracking_ok;
731
732 FormatProportionalFaults(snap, res.apply_errors, valve.GetLastError(),
733 res, res.fault_summary);
734
735 ESP_LOGI(kTag,
736 "[PROP %.*s] result: %s apply_errs=%lu fc=%lu "
737 "tracking[outliers=%lu/%lu max|err|=%d mA mean=%+d mA "
738 "↑mono=%d ↓mono=%d]",

757 //---------------------------------------------------------------------
758 // Parallel stress test: drive every TLE92466ED channel simultaneously
759 //---------------------------------------------------------------------
760 static ParallelStressPatternResult
761 RunParallelStress(ManifoldValves& system,
762 ManifoldValvesDiagnosticsThread& diag,
763 const SelfTestPlan& plan,
764 bool staggered) noexcept {
765 ParallelStressPatternResult result{};
766 result.executed = true;
767 for (std::size_t i = 0; i < kNumProportionals; ++i) {
768 result.channels[i].label = kProportionalMap[i].label;
769 result.channels[i].channel = kProportionalMap[i].manager_channel;
770 }
771
772 // Step ladder used for both patterns. "Triangle" so each channel
773 // visits every level twice and we get up + down dynamics. Values
774 // expressed as normalized [0..1] positions; the actual mA pushed
775 // is `pos × full_flow_current_ma` per channel (so VSO gets 0..66 mA
776 // while LM-Pro gets 0..115 mA — different stress per family).
777 constexpr std::array<float, 7> kLevels{
778 0.00f, 0.33f, 0.66f, 1.00f, 0.66f, 0.33f, 0.00f
779 };
780
781 const char* pattern_name = staggered ? "STAGGERED" : "UNIFORM";
782 ESP_LOGI(kTag, "");
783 ESP_LOGI(kTag, "════════════════════════════════════════════════════════════");
784 ESP_LOGI(kTag, " Parallel Stress Sweep — pattern=%s 6 channels driven simultaneously",
785 pattern_name);
786 ESP_LOGI(kTag, " per-step settle=%lu ms, steady-state=%lu ms, sample=%lu ms",
787 static_cast<unsigned long>(plan.parallel_settle_ms),
788 static_cast<unsigned long>(plan.parallel_steady_state_ms),
789 static_cast<unsigned long>(plan.parallel_sample_period_ms));
790 ESP_LOGI(kTag, "════════════════════════════════════════════════════════════");
791
792 // Snapshot per-channel apply-error / fault counters before we start
793 // so the stress test can report deltas instead of cumulative totals.
794 std::array<uint32_t, kNumProportionals> apply_errs_before{};
795 std::array<uint32_t, kNumProportionals> fault_count_before{};
796 for (std::size_t i = 0; i < kNumProportionals; ++i) {
797 auto& v = system.Get(static_cast<ProportionalId>(i));
798 apply_errs_before[i] = v.GetApplyErrorCount();
799 fault_count_before[i] = v.GetFaultCount();
800 }
801
802 const std::size_t num_levels = kLevels.size();
803 // For uniform: walk all levels once. For staggered: walk num_levels
804 // steps where channel i sees level[(step + i) % num_levels] — after
805 // one full rotation every channel has seen every level.
806 const std::size_t num_steps = num_levels;
807
808 for (std::size_t step = 0; step < num_steps; ++step) {
809 // ── 1) Push setpoints to every channel ─────────────────────
810 for (std::size_t i = 0; i < kNumProportionals; ++i) {
811 const float pos = staggered
812 ? kLevels[(step + i) % num_levels]
813 : kLevels[step];
814 auto& v = system.Get(static_cast<ProportionalId>(i));
815 v.RequestPosition(pos);
816 }
817
818 // ── 2) Settle wait ─────────────────────────────────────────
819 os_delay_msec(plan.parallel_settle_ms);
820 WaitForDiagPass(diag);
821
822 // ── 3) Steady-state characterization per channel ────────────
823 //
824 // Sample every `parallel_sample_period_ms` for the configured
825 // window. Per channel: accumulate sum and sum_of_squares for a
826 // running mean+stddev, plus track max |excursion|. Per
827 // subsystem: track the worst SubsystemFaultSnapshot flags seen.
828 std::array<int64_t, kNumProportionals> sum{};
829 std::array<int64_t, kNumProportionals> sum_sq{};
830 std::array<uint16_t, kNumProportionals> n{};
831 std::array<int16_t, kNumProportionals> max_excursion{};
832 std::array<bool, kNumProportionals> step_any_fault{};
833 std::array<bool, kNumProportionals> step_any_warn{};
834
835 // Cache the commanded mA per channel for this step.
836 std::array<int16_t, kNumProportionals> step_cmd{};
837 for (std::size_t i = 0; i < kNumProportionals; ++i) {
838 const float pos = staggered
839 ? kLevels[(step + i) % num_levels]
840 : kLevels[step];
841 step_cmd[i] = static_cast<int16_t>(
842 pos *
843 static_cast<float>(kProportionalMap[i].model.full_flow_current_ma) +
844 0.5f);
845 }
846
847 const uint32_t window_steps =
848 (plan.parallel_steady_state_ms + plan.parallel_sample_period_ms - 1U) /
849 plan.parallel_sample_period_ms;
850 for (uint32_t s = 0; s < window_steps; ++s) {
851 os_delay_msec(plan.parallel_sample_period_ms);
852 for (std::size_t i = 0; i < kNumProportionals; ++i) {
853 auto& v = system.Get(static_cast<ProportionalId>(i));
854 const auto snap = v.Snapshot();
855 const int16_t actual = snap.average_current_mA;
856 const int16_t err = static_cast<int16_t>(actual - step_cmd[i]);
857 const int16_t abs_e = static_cast<int16_t>(err < 0 ? -err : err);
858 sum[i] += actual;
859 sum_sq[i] += static_cast<int64_t>(actual) * actual;
860 if (abs_e > max_excursion[i]) max_excursion[i] = abs_e;
861 if (snap.any_fault()) step_any_fault[i] = true;
862 if (snap.any_warning()) step_any_warn[i] = true;
863 ++n[i];
864 }
865 // Subsystem-level scrape every sample for VBAT droop / OT.
866 const auto sub = diag.GetSnapshot();
867 if (sub.tle_vbat_uv) result.vbat_under_voltage_seen = true;
868 if (sub.tle_ot_warning) result.ot_warning_seen = true;
869 if (sub.tle_ot_error) result.ot_error_seen = true;
870 if (sub.tle_clock_fault) result.clock_fault_seen = true;
871 }
872
873 // ── 4) Log per-channel observation for this step ───────────
874 ESP_LOGI(kTag,
875 " step %u/%u pattern=%s",
876 static_cast<unsigned>(step + 1U),
877 static_cast<unsigned>(num_steps),
878 pattern_name);
879 for (std::size_t i = 0; i < kNumProportionals; ++i) {
880 ParallelChannelObservation obs{};
881 obs.cmd_mA = step_cmd[i];
882 obs.max_excursion_mA = max_excursion[i];
883 obs.any_fault = step_any_fault[i];
884 obs.any_warning = step_any_warn[i];
885 if (n[i] > 0) {
886 const int64_t mean = sum[i] / n[i];
887 obs.mean_actual_mA = static_cast<int16_t>(mean);
888 obs.mean_err_mA = static_cast<int16_t>(mean - step_cmd[i]);
889 // stddev = sqrt(E[X²] − E[X]²)
890 const int64_t var = (sum_sq[i] / n[i]) - (mean * mean);
891 const int32_t v32 = (var < 0) ? 0 : static_cast<int32_t>(var);
892 obs.stddev_mA = IsqrtClamped(v32);
893 }
894 // ── Evaluate per-step pass against tolerances ───────────
895 const float abs_cmd_f = static_cast<float>(
896 obs.cmd_mA < 0 ? -obs.cmd_mA : obs.cmd_mA);
897 const float band_err = std::max(
898 plan.parallel_max_abs_mean_err_min_mA,
899 (plan.parallel_max_abs_mean_err_pct * abs_cmd_f) / 100.0f);
900 const float band_exc = std::max(
901 plan.parallel_max_excursion_min_mA,
902 (plan.parallel_max_excursion_pct * abs_cmd_f) / 100.0f);
903
904 const int16_t abs_mean_err =
905 obs.mean_err_mA < 0 ? -obs.mean_err_mA : obs.mean_err_mA;
906
907 const bool ok_err = static_cast<float>(abs_mean_err) <= band_err;
908 const bool ok_sd = static_cast<float>(obs.stddev_mA) <= plan.parallel_max_stddev_mA;
909 const bool ok_exc = static_cast<float>(obs.max_excursion_mA) <= band_exc;
910 const bool ok_warn = !obs.any_warning ||
911 plan.parallel_allow_per_channel_warnings;
912 const bool step_ok = ok_err && ok_sd && ok_exc &&
913 !obs.any_fault && ok_warn;
914
915 ESP_LOGI(kTag,
916 " %-6.*s cmd=%4d mean=%4d err=%+4d σ=%3d "
917 "|exc|=%3d fault=%d warn=%d %s",
918 static_cast<int>(result.channels[i].label.size()),
919 result.channels[i].label.data(),
920 static_cast<int>(obs.cmd_mA),
921 static_cast<int>(obs.mean_actual_mA),
922 static_cast<int>(obs.mean_err_mA),
923 static_cast<int>(obs.stddev_mA),
924 static_cast<int>(obs.max_excursion_mA),
925 obs.any_fault ? 1 : 0,
926 obs.any_warning ? 1 : 0,
927 step_ok ? "OK" : "FAIL");
928
929 // ── Aggregate per channel ───────────────────────────────
930 auto& ch = result.channels[i];
931 if (abs_mean_err > ch.worst_mean_err_mA)
932 ch.worst_mean_err_mA = abs_mean_err;
933 if (obs.stddev_mA > ch.worst_stddev_mA)
934 ch.worst_stddev_mA = obs.stddev_mA;
935 if (obs.max_excursion_mA > ch.worst_excursion_mA)
936 ch.worst_excursion_mA = obs.max_excursion_mA;
937 if (obs.any_fault) ch.any_fault_seen = true;
938 if (obs.any_warning) ch.any_warning_seen = true;
939 if (!step_ok) ++ch.steps_failed;
940 }
941 }
942
943 // ── Wind down: drive everything back to 0 ──────────────────────
944 for (std::size_t i = 0; i < kNumProportionals; ++i) {
945 system.Get(static_cast<ProportionalId>(i)).RequestPosition(0.0f);
946 }
947 os_delay_msec(plan.parallel_settle_ms);
948 WaitForDiagPass(diag);
949
950 // ── Snapshot delta per-channel apply-error / fault counters ────
951 for (std::size_t i = 0; i < kNumProportionals; ++i) {
952 auto& v = system.Get(static_cast<ProportionalId>(i));
953 result.channels[i].apply_errors =
954 v.GetApplyErrorCount() - apply_errs_before[i];
955 result.channels[i].fault_count =
956 v.GetFaultCount() - fault_count_before[i];
957 }
958
959 result.worst_diag_iterations = static_cast<uint16_t>(diag.IterationCount());
960
961 // ── Roll up per-channel pass + pattern pass ────────────────────
962 result.pass = true;
963 std::array<char, 96> first_failure{};
964 bool have_first_failure = false;
965
966 // Subsystem-level fail conditions.
967 const auto note_subsystem_fail = [&](const char* what) {
968 if (!have_first_failure) {
969 std::snprintf(result.fail_reason.data(), result.fail_reason.size(),
970 "subsystem: %s", what);
971 have_first_failure = true;
972 }
973 result.pass = false;
974 };
975 if (plan.parallel_fail_on_vbat_uv && result.vbat_under_voltage_seen)
976 note_subsystem_fail("vbat_uv seen");
977 if (plan.parallel_fail_on_ot_warn && result.ot_warning_seen)
978 note_subsystem_fail("ot_warning seen");
979 if (plan.parallel_fail_on_ot_err && result.ot_error_seen)
980 note_subsystem_fail("ot_error seen");
981 if (plan.parallel_fail_on_clock_fault && result.clock_fault_seen)
982 note_subsystem_fail("clock_fault seen");
983
984 // Per-channel rollup.
985 for (std::size_t i = 0; i < kNumProportionals; ++i) {
986 auto& ch = result.channels[i];
987 ch.pass = true;
988 // Forbidden warnings / hard faults / apply errors.
989 const bool warn_blocks =
990 ch.any_warning_seen && !plan.parallel_allow_per_channel_warnings;
991 if (ch.any_fault_seen) { ch.pass = false; }
992 if (ch.apply_errors > 0) { ch.pass = false; }
993 if (ch.fault_count > 0) { ch.pass = false; }
994 if (warn_blocks) { ch.pass = false; }
995 if (ch.steps_failed > plan.parallel_max_failed_steps_per_channel) {
996 ch.pass = false;
997 }
998 if (!ch.pass) {
999 std::snprintf(ch.fail_reason.data(), ch.fail_reason.size(),
1000 "fail_steps=%lu/%lu fault=%d warn=%d apply_err=%lu fc=%lu",
1001 static_cast<unsigned long>(ch.steps_failed),
1002 static_cast<unsigned long>(num_steps),
1003 ch.any_fault_seen ? 1 : 0,
1004 ch.any_warning_seen ? 1 : 0,
1005 static_cast<unsigned long>(ch.apply_errors),
1006 static_cast<unsigned long>(ch.fault_count));
1007 if (!have_first_failure) {
1008 std::snprintf(result.fail_reason.data(), result.fail_reason.size(),
1009 "%.*s: %s",
1010 static_cast<int>(ch.label.size()), ch.label.data(),
1011 ch.fail_reason.data());
1012 have_first_failure = true;
1013 }
1014 result.pass = false;
1015 }
1016 }
1017
1018 ESP_LOGI(kTag, " pattern=%s complete — verdict=%s%s%s",
1019 pattern_name,
1020 result.pass ? "PASS" : "FAIL",
1021 result.pass ? "" : " (",
1022 result.pass ? "" : result.fail_reason.data());
1023 if (!result.pass) ESP_LOGE(kTag, " ↑ pattern=%s FAILED", pattern_name);
1024 ESP_LOGI(kTag, "");
1025 return result;
1026 }

1029 static int16_t IsqrtClamped(int32_t v) noexcept {
1030 if (v <= 0) return 0;
1031 // Newton's method, ~10 iterations is more than enough for int32_t.
1032 int32_t x = v;
1033 int32_t y = (x + 1) / 2;
1034 while (y < x) {
1035 x = y;
1036 y = (x + v / x) / 2;
1037 }
1038 return (x > INT16_MAX) ? INT16_MAX : static_cast<int16_t>(x);
1039 }
