181 uint32_t Step() noexcept override {
182 const uint64_t now_ms = static_cast<uint64_t>(os_get_elapsed_time_msec());
183 ValveManager& mgr = FLUX_API.valves;
184 const bool mgr_ready = mgr.IsInitialized();
185 manager_ready_.store(mgr_ready, std::memory_order_release);
186
187 if (mgr_ready) {
188 // ── Per-valve fault aggregation ───────────────────────────────
189 system_.ForEachSolenoid([this, now_ms](SolenoidValve& v) {
190 v.SetDiagnosticTimestamp(now_ms);
191 (void)v.OnDiagnostics();
192 });
193 system_.ForEachProportional([this, now_ms](ProportionalValve& v) {
194 v.SetDiagnosticTimestamp(now_ms);
195 (void)v.OnDiagnostics();
196 });
197
198 // ── System totals (sticky counters) ───────────────────────────
199 uint32_t total_f = 0, total_e = 0;
200 system_.ForEachValve([&](Valve& v) {
201 total_f += v.GetFaultCount();
202 total_e += v.GetApplyErrorCount();
203 });
204 total_faults_.store(total_f, std::memory_order_release);
205 total_apply_errors_.store(total_e, std::memory_order_release);
206
207 // ── MAX22200 global STATUS scrape ─────────────────────────────
208 ScrapeMax22200(mgr);
209 // ── TLE92466ED global FaultReport scrape ──────────────────────
210 ScrapeTle92466(mgr);
211 } else {
212 max22200_ready_.store(false, std::memory_order_release);
213 tle92466_ready_.store(false, std::memory_order_release);
214 max_flags_.store(0, std::memory_order_release);
215 tle_flags_.store(0, std::memory_order_release);
216 }
217
218 ads7952_ready_.store(ads7952_ != nullptr, std::memory_order_release);
219 last_update_ms_.store(now_ms, std::memory_order_release);
220 iteration_.fetch_add(1, std::memory_order_acq_rel);
221 return period_ms_;
222 }

224 bool Cleanup() noexcept override { return true; }

262 void ScrapeMax22200(ValveManager& mgr) noexcept {
263 Max22200Handler* h = mgr.GetMax22200Handler();
264 if (!h) {
265 max22200_ready_.store(false, std::memory_order_release);
266 max_flags_.store(0, std::memory_order_release);
267 return;
268 }
269 max22200::StatusConfig st{};
270 const auto rs = h->GetStatus(st);
271 if (rs != max22200::DriverStatus::OK) {
272 max22200_ready_.store(false, std::memory_order_release);
273 return;
274 }
275 max22200_ready_.store(true, std::memory_order_release);
276 uint32_t bits = 0;
277 if (st.overtemperature) bits |= kMaxOvt;
278 if (st.communication_error) bits |= kMaxComer;
279 if (st.undervoltage) bits |= kMaxUvm;
280 if (st.overcurrent) bits |= kMaxOcp;
281 if (st.open_load_fault) bits |= kMaxOlf;
282 if (st.hit_not_reached) bits |= kMaxHhf;
283 if (st.plunger_movement_fault) bits |= kMaxDpm;
284 if (st.active) bits |= kMaxActive;
285 max_flags_.store(bits, std::memory_order_release);
286 max_channels_on_mask_.store(st.channels_on_mask, std::memory_order_release);
287 }

289 void ScrapeTle92466(ValveManager& mgr) noexcept {
290 tle92466ed::FaultReport report{};
291 const hf_valve_err_t err = mgr.GetProportionalFaults(report);
292 if (err != hf_valve_err_t::VALVE_SUCCESS) {
293 tle92466_ready_.store(false, std::memory_order_release);
294 return;
295 }
296 tle92466_ready_.store(true, std::memory_order_release);
297 uint32_t bits = 0;
298 if (report.vbat_uv) bits |= kTleVbatUv;
299 if (report.vbat_ov) bits |= kTleVbatOv;
300 if (report.vio_uv) bits |= kTleVioUv;
301 if (report.vio_ov) bits |= kTleVioOv;
302 if (report.vdd_uv) bits |= kTleVddUv;
303 if (report.vdd_ov) bits |= kTleVddOv;
304 if (report.ot_warning) bits |= kTleOtWarn;
305 if (report.ot_error) bits |= kTleOtErr;
306 if (report.clock_fault) bits |= kTleClk;
307 if (report.spi_wd_error) bits |= kTleSpiWd;
308 if (report.por_event) bits |= kTlePor;
309 if (report.reset_event) bits |= kTleReset;
310 if (report.reg_ecc_err) bits |= kTleRegEcc;
311 if (report.otp_ecc_err) bits |= kTleOtpEcc;
312 if (report.supply_nok_internal) bits |= kTleSupNokInt;
313 if (report.supply_nok_external) bits |= kTleSupNokExt;
314 tle_flags_.store(bits, std::memory_order_release);
315 // Optionally also pull live VBAT/VIO from DeviceStatus (separate
316 // call) — the FaultReport does not include voltages.
317 Tle92466edHandler* h = mgr.GetTle92466Handler();
318 if (h) {
319 tle92466ed::DeviceStatus dev{};
320 // GetDeviceStatus exists on the underlying driver but the
321 // handler does not currently expose it. Fall through silently
322 // — the per-channel ChannelDiagnostics already publish
323 // average_current / vbat_feedback on every proportional valve.
324 (void)dev;
325 }
326 }
