99 hf_valve_err_t ApplyChannelFeatures() noexcept override {
100 ValveManager& mgr = FLUX_API.valves;
101 if (!IsMapped()) return hf_valve_err_t::VALVE_ERR_DRIVER_NOT_AVAILABLE;
102 if (features_pushed_.load(std::memory_order_acquire)) {
103 return hf_valve_err_t::VALVE_SUCCESS;
104 }
105
106 Tle92466edHandler* h = mgr.GetTle92466Handler();
107 if (!h) return hf_valve_err_t::VALVE_ERR_DRIVER_NOT_AVAILABLE;
108
109 const uint8_t hw = static_cast<uint8_t>(entry_.manager_channel) -
110 static_cast<uint8_t>(ValveChannel::PROP_CH0);
111
112 // ── Drop into Configuration Mode ──────────────────────────────────
113 // The TLE92466ED only accepts CH_x_CTRL writes (mode / slew / diag /
114 // open-load / dither / PWM period) while the chip is in Configuration
115 // Mode. The handler enters Mission Mode at init time so we have to
116 // round-trip CONFIG → write everything → MISSION here. Without this
117 // every ConfigureChannel call comes back with
118 // `OperationNotAllowedInMode`.
119 if (auto rc = h->EnterConfigMode(); !rc.has_value()) {
120 return TleDriverErrorToValveError(rc.error());
121 }
122
123 // Build a full ChannelConfig from the model spec and feature gates.
124 tle92466ed::ChannelConfig cfg{};
125 cfg.mode = tle92466ed::ChannelMode::ICC;
126 cfg.current_setpoint_ma = 0;
127 cfg.slew_rate = tle92466ed::SlewRate::MEDIUM_2V5_US;
128 cfg.diag_current = tle92466ed::DiagCurrent::I_80UA;
129 cfg.open_load_threshold = entry_.features.open_load_threshold;
130 cfg.olsg_warning_enabled = entry_.features.olsg_warning_enabled;
131 cfg.auto_limit_disabled = entry_.features.auto_limit_disabled;
132 cfg.deep_dither_enabled = entry_.features.deep_dither_enabled;
133 // Dither step / step-size / flat are filled by ValveManager when
134 // ConfigureProportionalDither is called below; leave them zeroed here.
135
136 const auto cfg_res = h->ConfigureChannel(hw, cfg);
137 if (!cfg_res.has_value()) {
138 (void)h->EnterMissionMode(); // restore state before returning
139 return TleDriverErrorToValveError(cfg_res.error());
140 }
141
142 // Push PWM frequency through the manager (it converts Hz → mantissa+exp).
143 const auto pwm_err = mgr.ConfigureProportionalPwm(
144 entry_.manager_channel, static_cast<float>(entry_.model.pwm_hz));
145 if (pwm_err != hf_valve_err_t::VALVE_SUCCESS) {
146 (void)h->EnterMissionMode();
147 return pwm_err;
148 }
149
150 // Optional dither (VSO-class only). Dither setup also writes
151 // CH_x_CTRL bits + DITHER_CLK_DIV — must run in Configuration Mode.
152 if (entry_.model.supports_dither &&
153 entry_.model.dither_amplitude_ma > 0 &&
154 entry_.model.dither_frequency_hz > 0) {
155 const auto d_err = mgr.ConfigureProportionalDither(
156 entry_.manager_channel,
157 static_cast<float>(entry_.model.dither_amplitude_ma),
158 static_cast<float>(entry_.model.dither_frequency_hz));
159 // Dither failure is non-fatal — keep going, just record the error.
160 if (d_err != hf_valve_err_t::VALVE_SUCCESS) {
161 RecordApplyResult(d_err, false);
162 }
163 }
164
165 // ── Re-enter Mission Mode so EnableChannel / SetChannelCurrent work ─
166 if (auto rc = h->EnterMissionMode(); !rc.has_value()) {
167 return TleDriverErrorToValveError(rc.error());
168 }
169
170 features_pushed_.store(true, std::memory_order_release);
171 configured_.store(true, std::memory_order_release);
172 return hf_valve_err_t::VALVE_SUCCESS;
173 }

175 hf_valve_err_t Apply() noexcept override {
176 ValveManager& mgr = FLUX_API.valves;
177 if (!IsMapped()) {
178 const auto err = hf_valve_err_t::VALVE_ERR_DRIVER_NOT_AVAILABLE;
179 SetState(ValveState::Uninitialized);
180 RecordApplyResult(err, false);
181 return err;
182 }
183
184 if (!features_pushed_.load(std::memory_order_acquire)) {
185 const auto fe = ApplyChannelFeatures();
186 if (fe != hf_valve_err_t::VALVE_SUCCESS) {
187 RecordApplyResult(fe, false);
188 return fe;
189 }
190 }
191
192 const float pos = GetTargetPosition();
193 const uint16_t cur_ma = PositionToCurrentMa(pos);
194
195 if (cur_ma == 0) {
196 const auto err = mgr.DisableChannel(entry_.manager_channel);
197 const bool ok = (err == hf_valve_err_t::VALVE_SUCCESS);
198 if (ok) {
199 SetActualPosition(0.0f);
200 SetState(ValveState::Idle);
201 }
202 RecordApplyResult(err, ok);
203 return err;
204 }
205
206 const auto enable_err = mgr.EnableChannel(entry_.manager_channel);
207 if (enable_err != hf_valve_err_t::VALVE_SUCCESS) {
208 RecordApplyResult(enable_err, false);
209 return enable_err;
210 }
211
212 const auto set_err = mgr.SetProportionalCurrent(entry_.manager_channel, cur_ma);
213 const bool ok = (set_err == hf_valve_err_t::VALVE_SUCCESS);
214 if (ok) {
215 const float applied = static_cast<float>(cur_ma) /
216 static_cast<float>(entry_.model.full_flow_current_ma);
217 SetActualPosition(applied > 1.0f ? 1.0f : applied);
218 SetState(ValveState::Active);
219 }
220 RecordApplyResult(set_err, ok);
221 return set_err;
222 }

228 hf_valve_err_t OnDiagnostics() noexcept override {
229 ValveManager& mgr = FLUX_API.valves;
230 if (!IsMapped()) {
231 diag_bits_.store(0, std::memory_order_release);
232 SetState(ValveState::Uninitialized);
233 return hf_valve_err_t::VALVE_ERR_DRIVER_NOT_AVAILABLE;
234 }
235
236 const uint8_t hw = static_cast<uint8_t>(entry_.manager_channel) -
237 static_cast<uint8_t>(ValveChannel::PROP_CH0);
238
239 // Per-channel rich diagnostics (current / duty / VBAT) come from the
240 // handler's GetChannelDiagnostics, not from FaultReport. Use both.
241 Tle92466edHandler* h = mgr.GetTle92466Handler();
242
243 bool any_fault = false;
244 uint32_t bits = kBitHwReady;
245
246 if (h) {
247 tle92466ed::ChannelDiagnostics ch_diag{};
248 const auto dr = h->GetChannelDiagnostics(hw, ch_diag);
249 if (dr.has_value()) {
250 if (ch_diag.overcurrent) { bits |= kBitOc; any_fault = true; }
251 if (ch_diag.short_to_ground) { bits |= kBitSg; any_fault = true; }
252 if (ch_diag.open_load) { bits |= kBitOl; any_fault = true; }
253 if (ch_diag.over_temperature) { bits |= kBitOt; any_fault = true; }
254 if (ch_diag.open_load_short_ground) { bits |= kBitOlSg; any_fault = true; }
255 if (ch_diag.ot_warning) bits |= kBitOtWarn;
256 if (ch_diag.current_regulation_warning) bits |= kBitCurRegWarn;
257 if (ch_diag.pwm_regulation_warning) bits |= kBitPwmRegWarn;
258 if (ch_diag.olsg_warning) bits |= kBitOlsgWarn;
259 avg_current_raw_.store(ch_diag.average_current, std::memory_order_release);
260 duty_cycle_raw_.store(ch_diag.duty_cycle, std::memory_order_release);
261 vbat_feedback_raw_.store(ch_diag.vbat_feedback, std::memory_order_release);
262 // Decode the raw FB_I_AVG / FB_DC mantissa+exponent pairs
263 // into engineering units. The driver's `GetAverageCurrent`
264 // / `GetDutyCycle` accessors do the same arithmetic, but
265 // re-issuing those calls would double the SPI cost. We
266 // already have the raw register values from the diagnostics
267 // read above, so just decode in-place using the same
268 // FB_FEEDBACK helpers used inside the driver.
269 const int32_t i_ma = tle92466ed::FB_FEEDBACK::ComputeAverageCurrent_mA(
270 ch_diag.average_current, ch_diag.duty_cycle);
271 const float dc = tle92466ed::FB_FEEDBACK::ComputeDutyCycle(ch_diag.duty_cycle);
272 const int32_t i_clamped =
273 (i_ma < INT16_MIN) ? INT16_MIN
274 : (i_ma > INT16_MAX ? INT16_MAX : i_ma);
275 const uint32_t dc_pm =
276 (dc <= 0.0f) ? 0U
277 : (dc >= 1.0f ? 10000U
278 : static_cast<uint32_t>(dc * 10000.0f + 0.5f));
279 avg_current_mA_.store(static_cast<int16_t>(i_clamped),
280 std::memory_order_release);
281 duty_cycle_permyriad_.store(static_cast<uint16_t>(dc_pm),
282 std::memory_order_release);
283 } else {
284 last_diag_err_ = TleDriverErrorToValveError(dr.error());
285 diag_bits_.store(0, std::memory_order_release);
286 return last_diag_err_;
287 }
288 } else {
289 // Handler unavailable — fall back to FaultReport via ValveManager.
290 tle92466ed::FaultReport report{};
291 const hf_valve_err_t err = mgr.GetProportionalFaults(report);
292 if (err != hf_valve_err_t::VALVE_SUCCESS) {
293 diag_bits_.store(0, std::memory_order_release);
294 last_diag_err_ = err;
295 return err;
296 }
297 const auto& ch = report.channels[hw];
298 if (ch.overcurrent) { bits |= kBitOc; any_fault = true; }
299 if (ch.short_to_ground) { bits |= kBitSg; any_fault = true; }
300 if (ch.open_load) { bits |= kBitOl; any_fault = true; }
301 if (ch.over_temperature) { bits |= kBitOt; any_fault = true; }
302 if (ch.open_load_short_ground) { bits |= kBitOlSg; any_fault = true; }
303 if (ch.ot_warning) bits |= kBitOtWarn;
304 if (ch.current_regulation_warning) bits |= kBitCurRegWarn;
305 if (ch.pwm_regulation_warning) bits |= kBitPwmRegWarn;
306 if (ch.olsg_warning) bits |= kBitOlsgWarn;
307 }
308
309 diag_bits_.store(bits, std::memory_order_release);
310 last_diag_err_ = hf_valve_err_t::VALVE_SUCCESS;
311
312 if (any_fault) {
313 diag_hw_fault_clear_streak_ = 0;
314 if (!diag_hw_fault_was_active_) {
315 RecordFault(hf_valve_err_t::VALVE_ERR_HARDWARE_FAULT, now_ms_);
316 }
317 diag_hw_fault_was_active_ = true;
318 return hf_valve_err_t::VALVE_ERR_HARDWARE_FAULT;
319 }
320 if (diag_hw_fault_was_active_) {
321 if (diag_hw_fault_clear_streak_ < UINT8_MAX) {
322 ++diag_hw_fault_clear_streak_;
323 }
324 if (diag_hw_fault_clear_streak_ >= kDiagHwFaultClearScrapes) {
325 diag_hw_fault_was_active_ = false;
326 diag_hw_fault_clear_streak_ = 0;
327 }
328 } else {
329 diag_hw_fault_clear_streak_ = 0;
330 }
331 return hf_valve_err_t::VALVE_SUCCESS;
332 }
