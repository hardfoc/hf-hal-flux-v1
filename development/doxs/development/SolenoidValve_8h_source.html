87 hf_valve_err_t ApplyChannelFeatures() noexcept override {
88 ValveManager& mgr = FLUX_API.valves;
89 if (features_pushed_.load(std::memory_order_acquire)) {
90 return hf_valve_err_t::VALVE_SUCCESS;
91 }
92 // The Flux::ValveManager doesn't expose per-feature setters yet,
93 // so reach through to the underlying handler. The handler caches
94 // its own ChannelConfig; fetching, mutating, and re-configuring is
95 // the safe round-trip.
96 Max22200Handler* h = mgr.GetMax22200Handler();
97 if (!h) return hf_valve_err_t::VALVE_ERR_DRIVER_NOT_AVAILABLE;
98
99 const uint8_t hw = static_cast<uint8_t>(entry_.manager_channel) -
100 static_cast<uint8_t>(ValveChannel::SOL_CH0);
101
102 // Build a fresh ChannelConfig from the model + per-channel feature
103 // gates. Drive mode is always CDR (current); side mode LOW_SIDE.
104 max22200::ChannelConfig cfg = (entry_.drive_mode == SolenoidDriveMode::BinaryHitHold)
105 ? max22200::ChannelConfig::makeSolenoidCdr(
106 static_cast<float>(entry_.model.hit_current_ma),
107 static_cast<float>(entry_.model.hold_current_ma),
108 entry_.model.min_hit_time_ms)
109 : max22200::ChannelConfig::makeSolenoidCdr(
110 /*hit_ma =*/ static_cast<float>(PositionToCurrentMa(GetTargetPosition())),
111 /*hold_ma =*/ static_cast<float>(PositionToCurrentMa(GetTargetPosition())),
112 /*hit_time=*/ 0.0f);
113
114 cfg.slew_rate_control_enabled = entry_.features.slew_rate_control;
115 cfg.open_load_detection_enabled = entry_.features.open_load_detection;
116 cfg.plunger_movement_detection_enabled = entry_.features.plunger_movement_detection;
117 cfg.hit_current_check_enabled = entry_.features.hit_current_check;
118
119 const auto err = MaxDriverStatusToValveError(h->ConfigureChannel(hw, cfg));
120 if (err == hf_valve_err_t::VALVE_SUCCESS) {
121 features_pushed_.store(true, std::memory_order_release);
122 configured_.store(true, std::memory_order_release);
123 }
124 return err;
125 }

127 hf_valve_err_t Apply() noexcept override {
128 ValveManager& mgr = FLUX_API.valves;
129 // Push channel features (and base CDR profile) on the very first
130 // apply, or after ForceReconfigure() / a fault recovery flow.
131 if (!features_pushed_.load(std::memory_order_acquire)) {
132 const auto fe = ApplyChannelFeatures();
133 if (fe != hf_valve_err_t::VALVE_SUCCESS) {
134 RecordApplyResult(fe, false);
135 return fe;
136 }
137 }
138
139 const float target = GetTargetPosition();
140
141 if (entry_.drive_mode == SolenoidDriveMode::BinaryHitHold) {
142 const bool want_on = (target > 0.0f);
143 const hf_valve_err_t err = want_on
144 ? mgr.EnableChannel(entry_.manager_channel)
145 : mgr.DisableChannel(entry_.manager_channel);
146 const bool ok = (err == hf_valve_err_t::VALVE_SUCCESS);
147 if (ok) {
148 SetActualPosition(want_on ? 1.0f : 0.0f);
149 SetState(want_on ? ValveState::Active : ValveState::Disabled);
150 }
151 RecordApplyResult(err, ok);
152 return err;
153 }
154
155 // ── VariableCurrent: re-issue ConfigureSolenoidCdr each apply ─────
156 const uint16_t target_ma = PositionToCurrentMa(target);
157 if (target_ma == 0) {
158 const auto err = mgr.DisableChannel(entry_.manager_channel);
159 const bool ok = (err == hf_valve_err_t::VALVE_SUCCESS);
160 if (ok) {
161 SetActualPosition(0.0f);
162 SetState(ValveState::Idle);
163 }
164 RecordApplyResult(err, ok);
165 return err;
166 }
167
168 // Push fresh CDR setpoints for this iteration.
169 const auto cfg_err = mgr.ConfigureSolenoidCdr(
170 entry_.manager_channel, target_ma, target_ma, /*hit_time_ms=*/0.0f);
171 if (cfg_err != hf_valve_err_t::VALVE_SUCCESS) {
172 RecordApplyResult(cfg_err, false);
173 return cfg_err;
174 }
175
176 const auto en_err = mgr.EnableChannel(entry_.manager_channel);
177 const bool ok = (en_err == hf_valve_err_t::VALVE_SUCCESS);
178 if (ok) {
179 const float applied = static_cast<float>(target_ma) /
180 static_cast<float>(entry_.model.full_flow_current_ma);
181 SetActualPosition(applied > 1.0f ? 1.0f : applied);
182 SetState(ValveState::Active);
183 }
184 RecordApplyResult(en_err, ok);
185 return en_err;
186 }

189 // Diagnostics: MAX22200 per-channel fault decoding

192 hf_valve_err_t OnDiagnostics() noexcept override {
193 ValveManager& mgr = FLUX_API.valves;
194 max22200::FaultStatus faults{};
195 const hf_valve_err_t err = mgr.GetSolenoidFaults(faults);
196 if (err != hf_valve_err_t::VALVE_SUCCESS) {
197 // Driver down; drop the cached snapshot to "uninitialized" so
198 // the application can react.
199 diag_bits_.store(0, std::memory_order_release);
200 last_diag_err_ = err;
201 return err;
202 }
203 last_diag_err_ = hf_valve_err_t::VALVE_SUCCESS;
204
205 const auto hw = static_cast<uint8_t>(entry_.manager_channel) -
206 static_cast<uint8_t>(ValveChannel::SOL_CH0);
207 const uint8_t ch_bit = static_cast<uint8_t>(1U << hw);
208
209 const bool ocp = (faults.overcurrent_channel_mask & ch_bit) != 0;
210 const bool hhf = (faults.hit_not_reached_channel_mask & ch_bit) != 0;
211 const bool olf = (faults.open_load_fault_channel_mask & ch_bit) != 0;
212 const bool dpm = (faults.plunger_movement_fault_channel_mask & ch_bit) != 0;
213
214 uint32_t bits = kBitHwReady;
215 if (ocp) bits |= kBitOcp;
216 if (hhf) bits |= kBitHhf;
217 if (olf) bits |= kBitOlf;
218 if (dpm) bits |= kBitDpm;
219 diag_bits_.store(bits, std::memory_order_release);
220
221 const bool hw_fault = (ocp || hhf || olf || dpm);
222 if (hw_fault) {
223 // Count fault *episodes* (rising edge), not every 10 Hz poll while
224 // the MAX22200 latched bits stay asserted — otherwise status
225 // `faults=` grows without bound during a single stuck condition.
226 diag_hw_fault_clear_streak_ = 0;
227 if (!diag_hw_fault_was_active_) {
228 RecordFault(hf_valve_err_t::VALVE_ERR_HARDWARE_FAULT, now_ms_);
229 }
230 diag_hw_fault_was_active_ = true;
231 return hf_valve_err_t::VALVE_ERR_HARDWARE_FAULT;
232 }
233 // Latched bits can read 0 for a scrape between SPI transactions; do not
234 // re-arm the edge detector until several consecutive clean reads.
235 if (diag_hw_fault_was_active_) {
236 if (diag_hw_fault_clear_streak_ < UINT8_MAX) {
237 ++diag_hw_fault_clear_streak_;
238 }
239 if (diag_hw_fault_clear_streak_ >= kDiagHwFaultClearScrapes) {
240 diag_hw_fault_was_active_ = false;
241 diag_hw_fault_clear_streak_ = 0;
242 }
243 } else {
244 diag_hw_fault_clear_streak_ = 0;
245 }
246 return hf_valve_err_t::VALVE_SUCCESS;
247 }

249 void SetDiagnosticTimestamp(uint64_t now_ms) noexcept { now_ms_ = now_ms; }
250 hf_valve_err_t GetLastDiagError() const noexcept { return last_diag_err_; }
